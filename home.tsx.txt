/** पहली स्क्रीन बन जाने के बाद ही चलाओ: ऐप तुरंत खुले, भारी काम बाद में */
function runWhenIdle(fn: () => void, timeout = 1500): () => void {
  if (typeof window === "undefined") return () => {};
  const w = window as any;
  if (typeof w.requestIdleCallback === "function") {
    const id = w.requestIdleCallback(fn, { timeout });
    return () => w.cancelIdleCallback?.(id);
  }
  const id = window.setTimeout(fn, 400);
  return () => window.clearTimeout(id);
}

/** सुरक्षा-घेरा: कोई एक स्क्रीन फ़ेल हो तो पूरा ऐप सफ़ेद न हो */
class SafeBoundary extends Component<
  { label: string; onReset?: () => void; children: ReactNode },
  { error: string | null }
> {
  state = { error: null as string | null };
  static getDerivedStateFromError(e: unknown) {
    return { error: e instanceof Error ? e.message : String(e) };
  }
  componentDidCatch(e: unknown) {
    console.error("[home] crash in", this.props.label, e);
  }
  render() {
    if (this.state.error !== null) {
      return (
        <div className="m-4 rounded-2xl border border-red-300 bg-red-50 p-4 text-center text-sm text-red-700 dark:bg-red-950/40 dark:text-red-200">
          <p className="font-semibold">{this.props.label} में दिक्कत आई</p>
          <p className="mt-1 break-words text-xs opacity-80">{this.state.error}</p>
          <button
            type="button"
            onClick={() => {
              this.setState({ error: null });
              this.props.onReset?.();
            }}
            className="mt-3 rounded-xl bg-red-600 px-4 py-2 text-xs font-bold text-white active:scale-95"
          >
            फिर से कोशिश करें
          </button>
        </div>
      );
    }
    return this.props.children;
  }
}

/** सेव की हुई लिस्ट को हमेशा {name, phone, avatar} रूप में पढ़ो (नेटिव सिंक में avatar नहीं होता) */
function normalizeContacts(parsed: unknown): PhoneContact[] {
  if (!Array.isArray(parsed)) return [];
  return (parsed as Record<string, unknown>[])
    .map((c) => {
      const phone = cleanNumber(String(c["phone"] ?? c["number"] ?? ""));
      const name = String(c["name"] ?? "").trim();
      const av = c["avatar"];
      return { name: name || phone, phone, avatar: typeof av === "string" && av ? av : "🙂" };
    })
    .filter((c) => c.phone.length === 10);
}

function loadContacts(): PhoneContact[] {
  try {
    const raw = localStorage.getItem(CONTACTS_KEY) ?? localStorage.getItem(OLD_CONTACTS_KEY);
    if (!raw) return [];
    return normalizeContacts(JSON.parse(raw));
  } catch {
    return [];
  }
}

/** सेव करते समय पुराने अतिरिक्त फ़ील्ड (जैसे onProChat) बने रहें */
function saveContacts(contacts: PhoneContact[]) {
  try {
    const prev = new Map<string, Record<string, unknown>>();
    try {
      const old = JSON.parse(localStorage.getItem(CONTACTS_KEY) || "[]") as Record<string, unknown>[];
      for (const c of Array.isArray(old) ? old : []) prev.set(cleanNumber(String(c["phone"] ?? "")), c);
    } catch {
      /* ignore */
    }
    localStorage.setItem(
      CONTACTS_KEY,
      JSON.stringify(contacts.map((c) => ({ ...(prev.get(c.phone) ?? {}), ...c }))),
    );
  } catch {
    /* storage full ya blocked */
  }
}

const ts = (iso?: string) => (iso ? Date.parse(iso) || 0 : 0);

function compactTime(iso?: string) {
  if (!iso) return "";
  const diffMs = Date.now() - ts(iso);
  const min = Math.floor(diffMs / 60000);
  if (min < 1) return "1S";
  if (min < 60) return `${min}M`;
  const hr = Math.floor(min / 60);
  if (hr < 24) return `${hr}H`;
  const day = Math.floor(hr / 24);
  if (day < 7) return `${day}D`;
  return `${Math.floor(day / 7)}W`;
}

/** चिप फ़िल्टर: "All" के अलावा बिना बातचीत वाले आइटम बाहर */
function inBucket(iso: string | undefined, chip: Chip) {
  if (chip === "All") return true;
  if (!iso) return false;
  const t = ts(iso);
  if (!t) return false;
  if (chip === "Today") return new Date(t).toDateString() === new Date().toDateString();
  const diffDays = (Date.now() - t) / 86_400_000;
  if (chip === "Weeks") return diffDays < 7;
  return diffDays < 30;
}

const SEEN_KEY = "pci_status_seen";
const DAY_MS = 86_400_000;

/** पिछले 24 घंटे में जिन नंबरों ने Status डाला (Supabase `statuses`): नंबर -> सबसे नए status का समय */
function useActiveStatuses() {
  const [map, setMap] = useState<Map<string, number>>(() => new Map());
  useEffect(() => {
    let dead = false;
    const load = async () => {
      try {
        const since = new Date(Date.now() - DAY_MS).toISOString();
        let res: any = await dbx.from("statuses").select("*").gte("created_at", since).limit(500);
        if (res?.error) res = await dbx.from("statuses").select("*").limit(300);
        if (dead || res?.error) return;
        const nowMs = Date.now();
        const next = new Map<string, number>();
        for (const r of (res?.data ?? []) as any[]) {
          const phone = cleanNumber(String(r?.owner_phone ?? r?.phone ?? r?.user_phone ?? r?.owner ?? ""));
          const created = Date.parse(String(r?.created_at ?? "")) || 0;
          const exp = r?.expires_at ? Date.parse(String(r.expires_at)) || 0 : 0;
          if (phone.length !== 10 || !created) continue;
          if (nowMs - created > DAY_MS) continue;
          if (exp && exp < nowMs) continue;
          next.set(phone, Math.max(next.get(phone) ?? 0, created));
        }
        setMap(next);
      } catch (e) {
        console.warn("status ring load failed:", e);
      }
    };

    let iv: number | undefined;
    let ch: ReturnType<typeof supabase.channel> | null = null;
    // रिंग पहले सेकंड में ज़रूरी नहीं: पहली स्क्रीन बनने के बाद शुरू
    const boot = window.setTimeout(() => {
      void load();
      iv = window.setInterval(load, 60_000);
      try {
        ch = supabase
          .channel(`home-status-${Math.random().toString(36).slice(2, 8)}`)
          .on("postgres_changes", { event: "INSERT", schema: "public", table: "statuses" }, () => void load())
          .subscribe();
      } catch {
        /* ignore */
      }
    }, 1800);

    return () => {
      dead = true;
      window.clearTimeout(boot);
      if (iv !== undefined) window.clearInterval(iv);
      if (ch) {
        try {
          supabase.removeChannel(ch);
        } catch {
          /* ignore */
        }
      }
    };
  }, []);
  return map;
}

/** किस नंबर का status कब तक देखा जा चुका है (localStorage) */
function useSeenStatuses() {
  const read = (): Record<string, number> => {
    try {
      const v = JSON.parse(localStorage.getItem(SEEN_KEY) || "{}");
      return v && typeof v === "object" ? (v as Record<string, number>) : {};
    } catch {
      return {};
    }
  };
  const [seen, setSeen] = useState<Record<string, number>>(read);
  const markSeen = useCallback((phone: string, upTo?: number) => {
    const p = cleanNumber(phone);
    if (!p) return;
    setSeen((cur) => {
      const next = { ...cur, [p]: Math.max(cur[p] ?? 0, upTo ?? Date.now()) };
      try {
        localStorage.setItem(SEEN_KEY, JSON.stringify(next));
      } catch {
        /* ignore */
      }
      return next;
    });
  }, []);
  // स्टेटस देखने पर window.dispatchEvent(new CustomEvent("pci-status-viewed", { detail: { phone } })) भेजा जाता है: रिंग ग्रे हो जाती है
  useEffect(() => {
    const onViewed = (e: Event) => {
      const phone = (e as CustomEvent<{ phone?: string }>).detail?.phone;
      if (phone) markSeen(phone);
    };
    window.addEventListener("pci-status-viewed", onViewed);
    return () => window.removeEventListener("pci-status-viewed", onViewed);
  }, [markSeen]);
  return { seen, markSeen };
}

/** किन नंबरों की Business प्रोफ़ाइल बनी है (Supabase `business_profiles`): इन्हीं से हुई चैट "Business" में दिखती हैं */
function useBusinessPhones() {
  const [phones, setPhones] = useState<Set<string>>(() => new Set());
  useEffect(() => {
    let dead = false;
    const stop = runWhenIdle(() => {
      void (async () => {
        try {
          const res: any = await dbx.from("business_profiles").select("*").limit(500);
          if (dead || res?.error) return;
          const next = new Set<string>();
          for (const r of (res?.data ?? []) as any[]) {
            const p = cleanNumber(String(r?.owner_phone ?? r?.phone ?? r?.user_phone ?? r?.owner ?? ""));
            if (p.length === 10) next.add(p);
          }
          setPhones(next);
        } catch (e) {
          console.warn("business phones load failed:", e);
        }
      })();
    }, 2500);
    return () => {
      dead = true;
      stop();
    };
  }, []);
  return phones;
}

function useRealChats(myPhone?: string) {
  const me = cleanNumber(myPhone);
  const [rows, setRows] = useState<ChatRow[]>([]);
  const [loading, setLoading] = useState(false);

  // पिछली बार की लिस्ट तुरंत दिखाओ (नेटवर्क का इंतज़ार नहीं)
  useEffect(() => {
    if (me.length !== 10) return;
    try {
      const v = JSON.parse(localStorage.getItem(`pci_chat_rows_${me}`) || "[]");
      if (Array.isArray(v) && v.length) setRows((cur) => (cur.length ? cur : (v as ChatRow[])));
    } catch {
      /* ignore */
    }
  }, [me]);

  useEffect(() => {
    if (me.length !== 10) {
      setRows([]);
      return;
    }
    let active = true;
    setLoading(true);

    async function fetchLatest(): Promise<any[] | null> {
      // तेज़ रास्ता: हर बातचीत का सिर्फ़ आख़िरी मैसेज (SQL फ़ंक्शन pci_recent_chats)
      try {
        const rpc = await dbx.rpc("pci_recent_chats", { p_me: me, p_limit: 150 });
        if (!rpc.error && Array.isArray(rpc.data)) return rpc.data as any[];
      } catch {
        /* फ़ंक्शन नहीं बना */
      }
      // पुराना रास्ता, पर सीमित
      try {
        const { data, error } = await supabase
          .from("messages")
          .select("sender_phone, receiver_phone, text, created_at, delivered_at")
          .or(`sender_phone.eq.${me},receiver_phone.eq.${me}`)
          .order("created_at", { ascending: false })
          .limit(1500);
        if (error) {
          console.error("Supabase error:", error);
          return null;
        }
        return (data ?? []) as any[];
      } catch (e) {
        console.error("chats fetch failed:", e);
        return null; // ऑफ़लाइन: पिछली लिस्ट बनी रहे
      }
    }

    async function load() {
      const msgs = await fetchLatest();
      if (!active) return;
      if (msgs === null) {
        setLoading(false);
        return;
      }
      if (msgs.length === 0) {
        setRows([]);
        setLoading(false);
        return;
      }

      // ऐप खुला है: मुझे आए सारे अनदेखे मैसेज "Delivered" चिह्नित करो
      if (msgs.some((m) => cleanNumber(m.receiver_phone) === me && !m.delivered_at)) {
        const { error: dErr } = await supabase
          .from("messages")
          .update({ delivered_at: new Date().toISOString() })
          .eq("receiver_phone", me)
          .is("delivered_at", null);
        if (dErr) console.error("Supabase error:", dErr);
      }

      // हर बातचीत का सबसे नया मैसेज (msgs पहले से नए-से-पुराना क्रम में हैं)
      const latest = new Map<string, { text: string; created_at: string }>();
      for (const m of msgs) {
        const s = cleanNumber(m.sender_phone);
        const other = s === me ? cleanNumber(m.receiver_phone) : s;
        if (other && !latest.has(other)) latest.set(other, { text: m.text, created_at: m.created_at });
      }

      const phones = [...latest.keys()].slice(0, 150);
      const profiles = await queryProfiles((cols) =>
        dbx
          .from("profiles")
          .select(cols)
          .in(
            "phone",
            phones.flatMap((p) => [p, `+91${p}`, `91${p}`]),
          ),
      );

      const built: ChatRow[] = phones.map((phone) => {
        const p = profiles.find((x) => cleanNumber(x.phone) === phone);
        const l = latest.get(phone)!;
        return {
          name: profName(p) || phone,
          msg: previewLabel(l.text),
          time: compactTime(l.created_at),
          createdAt: l.created_at,
          unread: 0,
          avatar: profAvatar(p),
          phone,
        };
      });

      // सबसे नया मैसेज सबसे ऊपर
      built.sort((a, b) => ts(b.createdAt) - ts(a.createdAt));
      if (active) {
        setRows(built);
        setLoading(false);
        try {
          // base64 फ़ोटो cache में न डालें (localStorage भर जाता है)
          const slim = built.slice(0, 60).map((b) => (b.avatar.startsWith("data:") ? { ...b, avatar: "🙂" } : b));
          localStorage.setItem(`pci_chat_rows_${me}`, JSON.stringify(slim));
        } catch {
          /* ignore */
        }
      }
    }

    void load();

    // मैसेज झुंड में आएँ तो एक ही बार reload
    let timer = 0;
    const reloadSoon = () => {
      window.clearTimeout(timer);
      timer = window.setTimeout(() => void load(), 350);
    };
    const channel = supabase
      .channel(`home-conversations-${me}-${Math.random().toString(36).slice(2, 8)}`)
      .on("postgres_changes", { event: "INSERT", schema: "public", table: "messages" }, (payload) => {
        const m = payload.new as { sender_phone: string; receiver_phone: string };
        if (cleanNumber(m.sender_phone) === me || cleanNumber(m.receiver_phone) === me) reloadSoon();
      })
      .subscribe();
    window.addEventListener("pci-back-online", reloadSoon);

    return () => {
      active = false;
      window.clearTimeout(timer);
      window.removeEventListener("pci-back-online", reloadSoon);
      supabase.removeChannel(channel);
    };
  }, [me]);

  return { rows, loading };
}

/** मेरे असली ग्रुप (Supabase groups + group_members) और हर ग्रुप का आख़िरी मैसेज */
function useRealGroups(myPhone?: string) {
  const me = cleanNumber(myPhone);
  const [groupRows, setGroupRows] = useState<GroupRow[]>([]);
  const idsRef = useRef<Set<string>>(new Set());

  // पिछली बार की ग्रुप सूची तुरंत (ऑफ़लाइन में भी दिखे)
  useEffect(() => {
    if (me.length !== 10) return;
    try {
      const v = JSON.parse(localStorage.getItem(`pci_group_rows_${me}`) || "[]");
      if (Array.isArray(v) && v.length) {
        setGroupRows((cur) => (cur.length ? cur : (v as GroupRow[])));
        idsRef.current = new Set((v as GroupRow[]).map((g) => String(g.id)));
      }
    } catch {
      /* ignore */
    }
  }, [me]);

  const load = useCallback(async () => {
    if (me.length !== 10) {
      idsRef.current = new Set();
      setGroupRows([]);
      return;
    }
    try {
      const { data: mine, error } = await supabase
        .from("group_members")
        .select("group_id, groups(id, name, avatar, created_at)")
        .eq("phone", me);
      if (error) throw error;

      const list = (mine ?? []) as any[];
      const ids = list.map((r) => String(r.group_id)).filter(Boolean);
      idsRef.current = new Set(ids);
      if (ids.length === 0) {
        setGroupRows([]);
        try {
          localStorage.removeItem(`pci_group_rows_${me}`);
        } catch {
          /* ignore */
        }
        return;
      }

      const [memRes, msgRes] = await Promise.all([
        supabase.from("group_members").select("group_id, phone").in("group_id", ids),
        supabase
          .from("group_messages")
          .select("group_id, sender_phone, text, created_at")
          .in("group_id", ids)
          .order("created_at", { ascending: false })
          .limit(400);
      ]);

const membersBy = new Map<string, string[]>();
      for (const r of (memRes.data ?? []) as any[]) {
        const gid = String(r.group_id);
        const arr = membersBy.get(gid) ?? [];
        arr.push(cleanNumber(String(r.phone)));
        membersBy.set(gid, arr);
      }

      const lastBy = new Map<string, any>();
      for (const m of (msgRes.data ?? []) as any[]) {
        const gid = String(m.group_id);
        if (!lastBy.has(gid)) lastBy.set(gid, m);
      }

      const built: GroupRow[] = [];
      for (const r of list) {
        const g = Array.isArray(r.groups) ? r.groups[0] : r.groups;
        if (!g) continue;
        const gid = String(g.id);
        const last = lastBy.get(gid);
        built.push({
          id: gid,
          name: String(g.name ?? "Group"),
          avatar: String(g.avatar || "👥"),
          createdAt: String(g.created_at ?? ""),
          memberPhones: membersBy.get(gid) ?? [me],
          ...(last
            ? {
                lastText: String(last.text ?? ""),
                lastSender: cleanNumber(String(last.sender_phone ?? "")),
                lastAt: String(last.created_at ?? ""),
              }
            : {}),
        });
      }
      setGroupRows(built);
      try {
        const slim = built.map((g) => (g.avatar.startsWith("data:") ? { ...g, avatar: "👥" } : g));
        localStorage.setItem(`pci_group_rows_${me}`, JSON.stringify(slim));
      } catch {
        /* ignore */
      }
    } catch (e) {
      console.error("groups load failed:", e);
    }
  }, [me]);

  useEffect(() => {
    void load();
    if (me.length !== 10) return;

    const onChanged = () => void load();
    window.addEventListener("pci-groups-changed", onChanged);
    window.addEventListener("pci-back-online", onChanged);

    const channel = supabase
      .channel(`home-groups-${me}-${Math.random().toString(36).slice(2, 8)}`)
      .on("postgres_changes", { event: "INSERT", schema: "public", table: "group_members" }, (payload) => {
        const r = payload.new as { phone?: string };
        if (cleanNumber(r.phone ?? "") === me) void load();
      })
      .on("postgres_changes", { event: "INSERT", schema: "public", table: "group_messages" }, (payload) => {
        const m = payload.new as { group_id?: string };
        if (m.group_id && idsRef.current.has(String(m.group_id))) void load();
      })
      .subscribe();

    return () => {
      window.removeEventListener("pci-groups-changed", onChanged);
      window.removeEventListener("pci-back-online", onChanged);
      supabase.removeChannel(channel);
    };
  }, [load, me]);

  return { groupRows, reloadGroups: load };
}

                                setActiveChat(rowToContact(c));
                              } else {
                                setProfileCard({ name: c.name, phone: c.phone, avatar: c.avatar });
                              }
                            }}
                            className="shrink-0 transition-transform active:scale-95"
                          >
                            <StatusRing state={c.groupId ? "none" : ringOf(c.phone)}>
                              <AvatarFace avatar={c.avatar} />
                            </StatusRing>
                          </button>
                          <div className="min-w-0 flex-1">
                            <p className="truncate text-sm font-semibold">{c.name}</p>
                            <p className="truncate text-xs text-muted-foreground">{c.msg}</p>
                          </div>
                          <span className="flex h-6 shrink-0 items-center justify-center rounded-full bg-emerald-500 px-2.5 text-[10px] font-bold text-white">
                            {compactTime(c.createdAt) || c.time}
                          </span>
                        </div>
                      ))}
                    </div>
                  ) : null}

                  {nothingFound ? (
                    <div className="space-y-3 py-8 text-center">
                      <p className="text-sm text-muted-foreground">No matching contacts.</p>
                      {queryDigits.length >= 10 ? (
                        <button
                          type="button"
                          onClick={() => startNewChat(query)}
                          className="rounded-2xl bg-emerald-500 px-5 py-2.5 text-sm font-semibold text-white active:scale-95"
                        >
                          Start chat with "{queryDigits.slice(-10)}"
                        </button>
                      ) : null}
                    </div>
                  ) : null}

                  {!isSearching &&
                  chip !== "All" &&
                  sortedPhoneContacts.length === 0 &&
                  recentChats.length === 0 ? (
                    <p className="px-6 py-14 text-center text-sm text-muted-foreground">
                      इस फ़िल्टर ({chip}) में कोई बातचीत नहीं है।
                    </p>
                  ) : null}

                  {!isSearching &&
                  chip === "All" &&
                  contactsLoaded &&
                  !chatsLoading &&
                  !syncingContacts &&
                  sortedPhoneContacts.length === 0 &&
                  recentChats.length === 0 ? (
                    <div className="flex flex-col items-center gap-3 px-6 py-16 text-center">
                      <MessageCircleOff className="size-10 text-muted-foreground/60" />
                      <p className="text-sm text-muted-foreground">
                        No contacts yet. Sync your contacts or search a number to start chatting.
                      </p>
                      <button
                        type="button"
                        onClick={handleSyncContacts}
                        className="rounded-2xl bg-emerald-500 px-5 py-2.5 text-sm font-semibold text-white active:scale-95"
                      >
                        Sync Contacts
                      </button>
                    </div>
                  ) : null}
                </div>
              ) : null}

              {tab === "groups" ? (
                <div className="px-3 pb-28 pt-3 md:px-4 lg:pb-6">
                  {/* Groups | Business पिल-टॉगल (Status | Business Showcase जैसा) */}
                  <div className="mx-auto mb-3 flex w-full max-w-xs items-center rounded-full bg-secondary/40 p-1">
                    <button
                      type="button"
                      onClick={() => setGroupSide("groups")}
                      className={cn(
                        "flex-1 rounded-full py-2 text-xs font-semibold transition-all",
                        groupSide === "groups"
                          ? "bg-gradient-to-r from-indigo-500 to-blue-600 text-white shadow-sm"
                          : "text-muted-foreground",
                      )}
                    >
                      Groups
                    </button>
                    <button
                      type="button"
                      onClick={selectBusinessSide}
                      className={cn(
                        "flex-1 rounded-full py-2 text-xs font-semibold transition-all",
                        groupSide === "business"
                          ? "bg-gradient-to-r from-purple-500 to-pink-500 text-white shadow-sm"
                          : "text-muted-foreground",
                      )}
                    >
                      Business
                    </button>
                  </div>

                  {groupSide === "groups" ? (
                    <>
                      <button
                        type="button"
                        onClick={() => setGroupOpen(true)}
                        className="mb-3 flex w-full items-center gap-3 rounded-full bg-white px-4 py-3 text-start shadow-sm dark:bg-zinc-800"
                      >
                        <span className="flex size-11 shrink-0 items-center justify-center rounded-full bg-brand-gradient text-white shadow-glow">
                          <Users className="size-5" />
                        </span>
                        <span className="text-sm font-semibold">+ New Group</span>
                      </button>

                      {groupChats.length > 0 ? (
                        <div className="space-y-2">
                          {[...groupChats]
                            .sort((a, b) => ts(b.createdAt) - ts(a.createdAt))
                            .map((g) => (
                              <button
                                key={g.groupId ?? g.name}
                                type="button"
                                onClick={() => setActiveChat(rowToContact(g))}
                                className="flex w-full items-center gap-3 rounded-[28px] bg-white p-3 text-start shadow-sm transition-transform active:scale-[0.99] dark:bg-zinc-800"
                              >
                                <StatusRing>
                                  <AvatarFace avatar={g.avatar || "👥"} />
                                </StatusRing>
                                <div className="min-w-0 flex-1">
                                  <p className="truncate text-sm font-semibold">{g.name}</p>
                                  <p className="truncate text-xs text-muted-foreground">
                                    {(g.memberPhones ?? []).length} members · {g.msg}
                                  </p>
                                </div>
                                <span className="flex h-6 shrink-0 items-center justify-center rounded-full bg-emerald-500 px-2.5 text-[10px] font-bold text-white">
                                  {compactTime(g.createdAt) || g.time}
                                </span>
                              </button>
                            ))}
                        </div>
                      ) : (
                        <div className="flex flex-col items-center gap-3 px-6 py-16 text-center">
                          <Users className="size-10 text-muted-foreground/60" />
                          <p className="text-sm text-muted-foreground">अभी आप किसी ग्रुप में नहीं हैं। नया ग्रुप बनाएँ।</p>
                        </div>
                      )}
                    </>
                  ) : (
                    <>
                      {/* Business: पतली पट्टी; पूरी सेटिंग "मेरा Business" बटन में */}
                      <div className="mb-3 flex items-center gap-2 rounded-2xl bg-gradient-to-r from-purple-500 via-fuchsia-500 to-pink-500 px-3 py-2 text-white shadow-md">
                        <BriefcaseBusiness className="size-5 shrink-0" />
                        <p className="min-w-0 flex-1 truncate text-sm font-extrabold">Business Chats</p>
                        <button
                          type="button"
                          onClick={() => setBizOpen(true)}
                          className="flex h-9 shrink-0 items-center gap-1.5 rounded-full bg-white px-3 text-xs font-bold text-purple-600 active:scale-95"
                        >
                          <Settings2 className="size-4" /> मेरा Business
                        </button>
                      </div>
                      {bizSetup?.offer?.on ? (
                        <div className="mb-3">
                          <OfferBannerCard offer={bizSetup.offer} name={bizSetup.name} />
                        </div>
                      ) : null}

                      {businessChats.length > 0 ? (
                        <div className="space-y-2">
                          {businessChats.map((c) => (
                            <button
                              key={c.phone ?? c.name}
                              type="button"
                              onClick={() => setActiveChat(rowToContact(c))}
                              className="flex w-full items-center gap-3 rounded-[28px] bg-white p-3 text-start shadow-sm transition-transform active:scale-[0.99] dark:bg-zinc-800"
                            >
                              <StatusRing state={ringOf(c.phone)}>
                                <AvatarFace avatar={c.avatar} />
                              </StatusRing>
                              <div className="min-w-0 flex-1">
                                <p className="flex items-center gap-1.5 truncate text-sm font-semibold">
                                  <span className="truncate">{c.name}</span>
                                  <span className="shrink-0 rounded-full bg-purple-100 px-1.5 py-0.5 text-[9px] font-bold text-purple-600 dark:bg-purple-500/20 dark:text-purple-300">
                                    Business
                                  </span>
                                </p>
                                <p className="truncate text-xs text-muted-foreground">{c.msg}</p>
                              </div>
                              <span className="flex h-6 shrink-0 items-center justify-center rounded-full bg-purple-500 px-2.5 text-[10px] font-bold text-white">
                                {compactTime(c.createdAt) || c.time}
                              </span>
                            </button>
                          ))}
                        </div>
                      ) : (
                        <div className="flex flex-col items-center gap-3 px-6 py-12 text-center">
                          <Store className="size-10 text-muted-foreground/60" />
                          <p className="text-sm text-muted-foreground">
                            अभी किसी दुकानदार से बातचीत नहीं है। Status › Business Showcase से किसी दुकान को चैट करें, या अपना Business बनाएँ।
                          </p>
                        </div>
                      )}
                    </>
                  )}
                </div>
              ) : null}

              {tab === "shopping" ? (
                <ShoppingTab
                  keypadStyle={state.theme.keypadStyle}
                  onOpenBusiness={() => {
                    setTab("groups");
                    selectBusinessSide();
                  }}
                />
              ) : null}

              {tab === "status" ? (
                <div className="space-y-4 pb-28 pt-4 lg:pb-6">
                  <div className="mx-auto flex w-full max-w-xs items-center rounded-full bg-secondary/40 p-1">
                    <button
                      type="button"
                      onClick={() => setStatusSubTab("status")}
                      className={cn(
                        "flex-1 rounded-full py-2 text-xs font-semibold transition-all",
                        statusSubTab === "status" ? "bg-emerald-500 text-white shadow-sm" : "text-muted-foreground",
                      )}
                    >
                      Status
                    </button>
                    <button
                      type="button"
                      onClick={() => setStatusSubTab("reels")}
                      className={cn(
                        "flex-1 rounded-full py-2 text-xs font-semibold transition-all",
                        statusSubTab === "reels"
                          ? "bg-gradient-to-r from-purple-500 to-pink-500 text-white shadow-sm"
                          : "text-muted-foreground",
                      )}
                    >
                      Business Showcase
                    </button>
                  </div>

    {statusSubTab === "status" ? (
                    <StatusPanel
                      myPhone={state.profile.phone}
                      contacts={phoneContacts}
                      defaultExpiry={state.storyExpiry ?? "24h"}
                      keypadStyle={state.theme.keypadStyle}
                      onChat={(phone) => openChatByPhone(phone)}
                    />
                  ) : null}
                </div>
              ) : null}

              {tab === "calls" ? (
                <CallsTab
                  myPhone={state.profile.phone}
                  contacts={phoneContacts}
                  keypadStyle={state.theme.keypadStyle}
                  onOpenChat={(phone) => openChatByPhone(phone)}
                  onCall={(phone, kind, name) =>
                    verifyAndCall(
                      { name: name || phone, phone: cleanNumber(phone), avatar: "📞" },
                      kind as CallKind,
                    )
                  }
                />
              ) : null}
            </SafeBoundary>
          </section>
        </div>

        {activeChat ? (
          <div
            style={appBgStyle}
            className={cn(
              "fixed inset-0 z-[9999] h-full w-full overflow-hidden lg:relative lg:z-auto lg:h-full lg:min-w-0 lg:flex-1",
              !hasAppBg && "bg-background",
            )}
          >
            <SafeBoundary key={activeChat.groupId ?? activeChat.phone ?? activeChat.name} label="चैट" onReset={() => setActiveChat(null)}>
              {activeChat.groupId ? (
                <GroupChatScreen
                  group={{
                    id: activeChat.groupId,
                    name: activeChat.name,
                    avatar: activeChat.avatar || "👥",
                    memberPhones: activeChat.memberPhones ?? [],
                  }}
                  myPhone={state.profile.phone}
                  nameOf={phoneName}
                  onBack={() => setActiveChat(null)}
                  keypadStyle={state.theme.keypadStyle}
                />
              ) : (
                <ChatScreen
                  contact={activeChat}
                  onBack={() => setActiveChat(null)}
                  keypadStyle={state.theme.keypadStyle}
                  wallpaper={state.theme.chatWallpaper ?? state.theme.appWallpaper}
                  t={t}
                  myPhone={state.profile.phone}
                  onContactUpdate={handleContactUpdate}
                  onOpenSettings={() => {
                    setActiveChat(null);
                    setSettingsOpen(true);
                  }}
                  sendReadReceipts={readReceiptsOn}
                  disappearAfterHours={disappearHours}
                  spamScamProtection={Boolean(privacy.scamProtection)}
                />
              )}
            </SafeBoundary>
          </div>
        ) : (
          <div className="hidden flex-1 items-center justify-center lg:flex">
            <div className="glass-card max-w-sm rounded-3xl p-8 text-center">
              <AppIcon variant={state.theme.appIcon} className="mx-auto size-14 text-2xl" />
              <p className="font-display mt-4 text-base font-semibold">Pro Chat India</p>
              <p className="mt-1 text-xs text-muted-foreground">
                Select a conversation to start messaging securely.
              </p>
            </div>
          </div>
        )}

        {!activeChat ? (
          <nav className="fixed bottom-3 inset-x-3 z-40 mx-auto w-auto max-w-sm lg:hidden">
            <div className="flex items-end justify-around rounded-[28px] border border-black/5 bg-white/90 px-1.5 pb-2 pt-2 shadow-2xl backdrop-blur-xl dark:border-white/10 dark:bg-zinc-900/90">
              {BOTTOM_NAV.map((item) => (
                <button
                  key={item.key}
                  type="button"
                  onClick={item.onClick}
                  aria-label={item.label}
                  aria-current={item.active ? "page" : undefined}
                  className="relative flex flex-1 flex-col items-center gap-0.5 px-1 transition-transform duration-300 ease-[cubic-bezier(0.34,1.56,0.64,1)] active:scale-90"
                >
                  <span
                    className={cn(
                      "flex size-12 items-center justify-center rounded-2xl transition-all duration-300 ease-[cubic-bezier(0.34,1.56,0.64,1)]",
                      item.active ? `${item.tint} -translate-y-1.5 scale-110 shadow-md` : "bg-transparent",
                    )}
                  >
                    <item.Icon className={cn("size-8 transition-all", item.active ? "" : "opacity-90")} />
                  </span>
                  <span
                    className={cn(
                      "text-[10px] leading-none",
                      item.active ? `font-extrabold ${item.labelCls}` : "font-medium text-zinc-500 dark:text-zinc-400",
                    )}
                  >
                    {item.label}
                  </span>
                </button>
              ))}
            </div>
          </nav>
        ) : null}
      </div>

      <SettingsSheet open={settingsOpen} onClose={() => setSettingsOpen(false)}>
        <SafeBoundary label="सेटिंग्स" onReset={() => setSettingsOpen(false)}>
          <SettingsTab state={state} update={update} t={t} profileSignal={profileSignal} />
        </SafeBoundary>
      </SettingsSheet>

      {/* मेरा Business: बैनर/सेटिंग्स का अपना बैंगनी-गुलाबी रूप */}
      <SettingsSheet
        open={bizOpen}
        onClose={() => setBizOpen(false)}
        title="My Business"
        headerClass="bg-gradient-to-r from-purple-500 to-pink-500"
        compact
      >
        <SafeBoundary label="बिज़नेस" onReset={() => setBizOpen(false)}>
          <BusinessPanel
            phone={myClean}
            setup={bizSetup}
            keypadStyle={state.theme.keypadStyle}
            onChange={setBizSetup}
            onEdit={() => setBizSetupOpen(true)}
          />
          <BusinessTab
            state={state}
            update={update}
            onOpenChat={(phone, name) => {
              setBizOpen(false);
              setActiveChat({ name, avatar: "🙂", msg: "", phone });
            }}
          />
        </SafeBoundary>
      </SettingsSheet>

      {tab === "status" && statusSubTab === "reels" ? (
        <SafeBoundary label="रील्स" onReset={() => setStatusSubTab("status")}>
          <ReelsFeed
            myPhone={state.profile.phone}
            onClose={() => setStatusSubTab("status")}
            onReply={(phone, item) => {
              setStatusSubTab("status");
              setTab("chats");
              openChatByPhone(phone, item.owner_name);
            }}
          />
        </SafeBoundary>
      ) : null}

      {liveCall ? (
        <div className="fixed inset-0 z-[70]">
          <CallScreen
            name={liveCall.name}
            avatar={liveCall.avatar}
            kind={liveCall.kind}
            myPhone={myClean}
            peerPhones={[liveCall.phone]}
            nameOf={phoneName}
            onEnd={() => setLiveCall(null)}
            ringingLabel={t("ringing")}
            endLabel={t("endCall")}
          />
        </div>
      ) : null}

      {chatTimerPromptOpen ? (
        <div className="fixed inset-0 z-[65] flex items-end justify-center bg-black/50 p-4 backdrop-blur-sm sm:items-center">
          <div className="glass-card animate-pop relative w-full max-w-xs rounded-3xl p-5 shadow-elevated">
            <button
              type="button"
              aria-label="Close"
              onClick={() => {
                // सिर्फ़ बंद करो; टाइमर अपने-आप चालू नहीं होना चाहिए (वरना मैसेज चुपचाप डिलीट होने लगेंगे)
                update({ disappearingPromptSeen: true });
                setChatTimerPromptOpen(false);
              }}
              className="absolute right-3 top-3 rounded-full p-1 text-muted-foreground transition-colors hover:text-foreground"
            >
              <X className="size-4" />
            </button>
            <p className="font-display pr-6 text-sm font-semibold">Set Message Timer</p>
            <p className="mt-2 text-xs leading-relaxed text-muted-foreground">
              Choose auto-delete duration for messages to save privacy and storage.
            </p>
            <button
              type="button"
              onClick={() => {
                update({ disappearingPromptSeen: true });
                setChatTimerPromptOpen(false);
                openTab("settings");
              }}
              className="bg-brand-gradient mt-4 w-full rounded-2xl py-2.5 text-sm font-semibold text-primary-foreground shadow-glow transition-all active:scale-95"
            >
              Set Duration
            </button>
          </div>
        </div>
      ) : null}

      {unregisteredCallPrompt ? (
        <div
          className="fixed inset-0 z-[60] flex items-center justify-center bg-black/50 p-6"
          onClick={() => setUnregisteredCallPrompt(null)}
        >
          <div
            className="w-full max-w-sm rounded-2xl bg-white p-5 text-center text-zinc-900 dark:bg-zinc-900 dark:text-white"
            onClick={(e) => e.stopPropagation()}
          >
            <p className="mb-2 text-3xl">📵</p>
            <p className="font-semibold">
              {unregisteredCallPrompt.name} अभी ProChat India पर नहीं है।
            </p>
            <p className="mt-1 text-xs text-muted-foreground">
              क्या आप सामान्य फ़ोन कॉल करना चाहते हैं या इनवाइट लिंक भेजना चाहते हैं?
            </p>
            <div className="mt-4 flex gap-2">
              <a
                href={`tel:+91${unregisteredCallPrompt.phone}`}
                className="flex-1 rounded-xl bg-emerald-500 px-3 py-2 text-sm font-semibold text-white"
              >
                Normal Call
              </a>
              <button
                onClick={async () => {
                  const text = `Pro Chat India पर मुझसे जुड़ें! यहाँ से डाउनलोड करें: https://digital-hindu-connect.vercel.app/get`;
                  try {
                    if (navigator.share) await navigator.share({ text });
                    else if (navigator.clipboard) await navigator.clipboard.writeText(text);
                  } catch {
                    /* cancelled */
                  }
                  setUnregisteredCallPrompt(null);
                }}
                className="flex-1 rounded-xl bg-secondary px-3 py-2 text-sm font-semibold"
              >
                Send Invite
              </button>
            </div>
            <button
              onClick={() => setUnregisteredCallPrompt(null)}
              className="mt-3 text-xs text-muted-foreground"
            >
              Cancel
            </button>
          </div>
        </div>
      ) : null}

      {showInvite ? (
        <div
          className="fixed inset-0 z-[60] flex items-center justify-center bg-black/50 p-6"
          onClick={() => setShowInvite(false)}
        >
          <div
            className="w-full max-w-sm rounded-2xl bg-white p-5 text-center text-zinc-900 dark:bg-zinc-900 dark:text-white"
            onClick={(e) => e.stopPropagation()}
          >
            <p className="mb-2 text-3xl">📵</p>
            <p className="font-semibold">
              This contact is not on Pro Chat India. Invite them to download the APK!
            </p>
            <button
              onClick={() => setShowInvite(false)}
              className="mt-4 rounded-xl bg-emerald-500 px-5 py-2 text-white"
            >
              OK
            </button>
          </div>
        </div>
      ) : null}

      {profileCard ? (
        <ContactCardModal
          card={profileCard}
          ring={ringOf(profileCard.phone)}
          onClose={() => setProfileCard(null)}
          onMessage={() => {
            const c = profileCard;
            setProfileCard(null);
            openChatByPhone(c.phone, c.name);
          }}
          onCall={(kind) => {
            const c = profileCard;
            setProfileCard(null);
            guard(() => verifyAndCall({ name: c.name, phone: c.phone, avatar: c.avatar }, kind))();
          }}
          onViewStatus={() => {
            const c = profileCard;
            markSeen(c.phone, statusMap.get(cleanNumber(c.phone)));
            setProfileCard(null);
            openTab("status");
          }}
        />
      ) : null}

      <BusinessSetupSheet
        open={bizSetupOpen}
        phone={myClean}
        initial={bizSetup}
        firstTime={!bizSetup}
        keypadStyle={state.theme.keypadStyle}
        onClose={() => setBizSetupOpen(false)}
        onSaved={(d) => {
          const first = !bizSetup;
          setBizSetup(d);
          setBizSetupOpen(false);
          if (first) setGroupSide("business");
        }}
      />

      <NewGroupDialog
        open={groupOpen}
        onOpenChange={setGroupOpen}
        contacts={phoneContacts.map((c) => ({ name: c.name, avatar: c.avatar, phone: c.phone }))}
        t={t}
        onCreate={(name, members) => void createGroup(name, members)}
      />

      {/* आने वाली कॉल: फ़ुल-स्क्रीन Accept / Decline */}
      <IncomingCallHost myPhone={state.profile.phone} nameOf={phoneName} />
      {/* दूसरे ऐप (YouTube/Chrome) से Share होकर आया लिंक: किसी चैट में भेजो */}
      <ShareTargetHost myPhone={state.profile.phone} />
      {modal}
      <BatteryGuide open={showBat} onClose={() => setShowBat(false)} />
    </main>
  );
}
